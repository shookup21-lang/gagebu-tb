rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // TBDR 가계부 (2026-09-19 강화)
    // - get만 허용: 문서 ID(=공유 암호에서 파생)를 아는 사람만 읽을 수 있다.
    //   list(목록 조회)를 막아서, 남이 문서 ID를 모아 암호를 추측하는 길을 차단한다.
    // - create/update만 허용, delete 금지: 암호를 모르는 사람이 지우지 못한다.
    // - version은 반드시 1씩 증가: 두 기기가 동시에 저장해도 늦은 쪽은 서버가 거부한다.
    match /ledgers/{docId} {
      allow get: if request.auth != null;

      allow create: if request.auth != null
                    && request.resource.data.version == 1;

      allow update: if request.auth != null
                    && request.resource.data.version == resource.data.version + 1;
    }

    // 그 외 모든 경로는 차단
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
